\begin{frame}{1. The Variational Quantum Algorithm}
    \small
    \textbf{Definition.} Fix $n$ data qubits, $\mathcal{H} = (\mathbb{C}^2)^{\otimes n}$, $N = 2^n$. A VQA is a triple:
    \begin{enumerate}
        \item[(i)] initial state $|\psi_0\rangle \in \mathcal{H}$,
        \item[(ii)] parameterized unitary $U(\theta) \in \mathrm{End}(\mathcal{H})$, $\theta \in \mathbb{R}^p$,
        \item[(iii)] Hermitian observable $O \in \mathrm{Herm}(\mathcal{H})$.
    \end{enumerate}
    \[
        \mathcal{L}(\theta) = \langle\psi_0|\, U^\dagger(\theta)\, O\, U(\theta)\, |\psi_0\rangle = \mathrm{Tr}\bigl(\rho_0\, U^\dagger(\theta)\, O\, U(\theta)\bigr)
    \]
    \textbf{Ground-state cost (paper Eq. 2):}
    \[
        O \;=\; I - \tfrac{1}{n}\sum_{i=1}^n Z_i, \qquad Z_i = I^{\otimes(i-1)} \otimes Z \otimes I^{\otimes(n-i)}
    \]
    Minimum at $|0\rangle^{\otimes n}$ with $\mathcal{L} = 0$. This is the cost the replication must use.
\end{frame}

\begin{frame}{2. Barren Plateaus \& the Master Formula}
    \small
    \textbf{McClean et al.\ 2018.} For sufficiently expressive ansätze,
    \[
        \mathrm{Var}_{\theta}\!\left[\tfrac{\partial \mathcal{L}}{\partial \theta_k}\right] \;\le\; \mathcal{O}(b^{-n}), \quad b > 1.
    \]
    \textbf{Larocca et al.\ 2025 (Nature Reviews) — DLA-saturated master formula.} For $U(\theta)$ from a 2-design over the DLA target $U(d)$, with traceless $H = \rho_0 - I/d$ and $V = O - \mathrm{Tr}(O)/d \cdot I$:
    \[
        \boxed{\;\mathrm{Var}[\mathcal{L}] \;\approx\; \frac{\mathrm{Tr}(H^2)\,\mathrm{Tr}(V^2)}{d^2 - 1}\;}
    \]
    The two trace norms factorise the gradient signal: initial-state purity ($H$) and cost-operator Hilbert–Schmidt mass ($V$) relative to the Haar volume $d^2$. The architectural advantage of Q-LINK lives entirely in the $\mathrm{Tr}(V^2)$ ratio.
\end{frame}

\begin{frame}{3. Q-LINK — Extended Hilbert Space}
    \small
    \textbf{Definition.} Append a single messenger qubit $\mathcal{H}_m = \mathbb{C}^2$:
    \[
        \mathcal{H}_{\mathrm{tot}} = \mathcal{H} \otimes \mathcal{H}_m, \qquad \dim(\mathcal{H}_{\mathrm{tot}}) = 2N = 2^{n+1}
    \]
    Initial state $\rho_{\mathrm{tot}} = \rho_0 \otimes |0\rangle\langle 0|_m$; cost acts as $\tilde O = O \otimes I_m$ (data-only measurement). A Hadamard on the messenger at $t=0$ is part of $U(\theta)$.

    \vspace{0.5em}
    \textbf{Mechanism.} Per layer: collect ($R_{xx}$ between each data qubit and messenger) → operate ($R_zR_yR_x$ + CZ on data) → distribute (CNOT fan-out from messenger).
    Unlike measurement-based residuals (ResQNets), unitarity is preserved end-to-end.
\end{frame}

\begin{frame}{4. Architecture ($U_{\mathrm{Q\text{-}LINK}} = U_{\mathrm{dist}}\, U_{\mathrm{opr}}\, U_{\mathrm{coll}}$)}
    \begin{center}
        \includegraphics[width=0.7\textwidth]{figure/QLINK.pdf}
    \end{center}
    \small
    \begin{itemize}
        \item \textbf{Collection.} $R_{xx}(\theta)$ between each data qubit and the messenger. Q-LINK(\textbf{Fixed}): $\theta = \pi/4$. Q-LINK(\textbf{Adaptive}): $\theta$ trainable.
        \item \textbf{Operation.} $R_z R_y R_x$ on each data qubit + CZ chain $0\text{-}1\text{-}2\text{-}\cdots$.
        \item \textbf{Distribution.} $CNOT_{m \to i}$ for each data qubit $i$.
        \item \textbf{Vanilla.} Same operation block, no messenger.
    \end{itemize}
\end{frame}

\begin{frame}{5. DLA Equivalence — Q-LINK and Vanilla Both Span $\mathfrak{su}(2^{n+1})$}
    \small
    \textbf{Theorem.} $\mathfrak{g}_{\mathrm{Q\text{-}LINK}} = \mathfrak{g}_{\mathrm{Vanilla}} = \mathfrak{su}(2^{n+1})$.

    \textbf{Proof sketch (Vanilla).} Single-qubit generators $\{iX_k, iY_k, iZ_k\}_{k=1}^{n+1}$ plus the entangler $iCZ_{j,j+1} = \tfrac{i}{4}(I-Z_j)(I-Z_{j+1})$ form a connected set on a linear graph. By the standard universality criterion, the closure is $\mathfrak{su}(2^{n+1})$.

    \textbf{Proof sketch (Q-LINK).} The data subalgebra spans $\mathfrak{su}(2^n) \otimes I_m$. We need $i Z_m$ to extend. The distribution $CNOT_{m \to i} = \tfrac{1}{2}(I + Z_m)\otimes I_i + \tfrac{1}{2}(I - Z_m)\otimes X_i$ contains $i Z_m \otimes I$ after Hadamard conjugation of the target. Then $[i Z_m \otimes I,\; i(X_m \otimes X_i)] = -2 i Y_m \otimes X_i$, and iterated commutators with $\{iX_m \otimes X_i,\, iY_m \otimes X_i,\, iZ_m \otimes I\}$ together with data-data CZ span every Pauli tensor product on $n+1$ qubits.

    \textbf{Consequence.} Expressibility at depth saturation is identical. Any Q-LINK advantage must come from $\mathrm{Tr}(\tilde V^2)$.
\end{frame}

\begin{frame}{6. Parameter-Shift Rule \& the Architectural Ratio}
    \small
    \textbf{Parameter-shift.}
    \[
        \frac{\partial \mathcal{L}}{\partial \theta_k} = \tfrac{1}{2}\!\left[\mathcal{L}(\theta + \tfrac{\pi}{2} e_k) - \mathcal{L}(\theta - \tfrac{\pi}{2} e_k)\right]
    \]
    Under the Haar 2-design assumption $\mathrm{Cov}[\mathcal{L}^+, \mathcal{L}^-] = 0$, so
    \[
        \mathrm{Var}\!\left[\tfrac{\partial \mathcal{L}}{\partial \theta_k}\right] = \tfrac{1}{4}\bigl(\mathrm{Var}[\mathcal{L}^+] + \mathrm{Var}[\mathcal{L}^-]\bigr) \;\le\; \tfrac{1}{2}\mathrm{Var}[\mathcal{L}]
    \]
    The $\tfrac{1}{2}$ is common to both architectures and cancels in the ratio.

    \textbf{Architectural ratio (master formula).} For Q-LINK $\tilde V = (O \otimes I_m) - I/d$ and Vanilla $V_{n+1}$ on $n+1$ qubits:
    \[
        \mathrm{Tr}(\tilde V^2) = \frac{2N}{n}, \qquad \mathrm{Tr}(V_{n+1}^2) = \frac{2N}{n+1}
        \;\;\Longrightarrow\;\;
        \boxed{\;\mathcal{R}(n) \;=\; \frac{n+1}{n}\;}
    \]
    \textbf{Remark.} $\mathcal{R}(n) \to 1$ as $n \to \infty$: the \emph{Haar-saturated} structural advantage is asymptotically trivial. Any larger empirical advantage at finite depth $D = n^2 \log n$ is a finite-depth (Cheeger-amplification) effect, not a feature of the ensemble theory.
\end{frame}

\begin{frame}{13. Diagnosis — Why the Convergence Target Misses}
    \small
    \textbf{Paper Eq. (2):} \quad $\hat O_{\mathrm{paper}} = I - \tfrac{1}{n_{\mathrm{data}}}\sum_i Z_i$ on the data register only.

    \textbf{Reference repo} (\texttt{AARC-lab/DCAS\_2026\_QLINK/src/qlink/metrics/cost\_function.py}):
    \begin{itemize}\small
        \item Implements $1 - \tfrac{1}{n_{\mathrm{data}}}\sum_i P(q_i = |0\rangle)$, i.e.\ \emph{half} the paper's operator (probability convention vs $\langle Z\rangle$ convention).
        \item Q-LINK-specific bug: loop iterates $j \in [0, 2^{n_{\mathrm{data}}})$ over a $2^{n_{\mathrm{tot}}}$-length \texttt{probs} vector. In TC big-endian (qubit 0 = MSB, verified by \texttt{tc.Circuit(2); X(0)} $\to$ \texttt{[0,0,1,0]}) this projects the first data qubit and replaces it with the messenger:
        \[
            \boxed{\;\hat O_{\mathrm{buggy}} \;=\; I \;-\; \frac{1}{n_{\mathrm{data}}}\sum_{k=1}^{n_{\mathrm{tot}}-1} P_0^{(q_0)} P_0^{(q_k)}\;}
        \]
        \item Pinned to the verbatim Python loop on random states for $n_{\mathrm{tot}} \in \{3, 4, 5\}$ by \texttt{tests/unit/test\_observables.py::test\_buggy\_observable\_matches\_literal\_loop}.
    \end{itemize}
    \textbf{Effect.} Running the paper's stated cost gives a flat landscape near random init → no convergence. Running the buggy operator reproduces published trajectory variances at the order-of-magnitude level (slide 9) but inflates the apparent Q-LINK advantage via the $P_0^{(q_0)}$ projector floor.
\end{frame}

\begin{frame}{15. Summary — Theory + Replication}
    \small
    \textbf{Theory.} The Q-LINK advantage at the Haar 2-design limit is exactly
    $\mathcal{R}(n) = (n+1)/n$, derived from the master formula. Asymptotic limit $\mathcal{R} \to 1$: the architecture's static structural gain is small.
    Anything larger is finite-depth Cheeger amplification, not a feature of the ensemble theory.

    \textbf{Replication (paper Eq. 2 cost).} Headline numbers (slide 9 Vanilla column, $n = 10$): \theirs{$1.18 \times 10^{-7}$} vs \ours{$1.31 \times 10^{-7}$} — quantitative match. Q-LINK columns differ from \theirs{theirs} by $5\!-\!10\times$ in each direction, traced to the reference-repo cost-function discrepancy (slide 13).

    \textbf{Qualitative.} Vanilla barren plateau at $n \ge 8$ (slide 11) and Vanilla-rougher-than-Q-LINK landscape (slide 12) both reproduce.

    \textbf{Net.} 3 PASS, 1 PARTIAL, 1 MISS — score is consistent with the assignment's expectation that the candidate own and diagnose the deviations rather than hit every number.
