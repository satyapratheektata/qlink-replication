% Slide 2: VQA and Cost Operator
\begin{frame}{The Variational Quantum Algorithm (VQA)}
    \textbf{Definition 1.1 (Data Hilbert Space and VQA).} Fix $n \in \mathbb{N}$ data qubits. The data Hilbert space is $\mathcal{H} = (\mathbb{C}^2)^{\otimes n}$, with $N = 2^n$. A VQA specifies:
    \begin{enumerate}
        \item[(i)] An initial state $|\psi_0\rangle \in \mathcal{H}$, prepared by an encoding circuit.
        \item[(ii)] A parameterized unitary $U(\theta) \in \mathrm{End}(\mathcal{H})$, $\theta \in \mathbb{R}^p$.
        \item[(iii)] An observable (cost operator) $O \in \mathrm{Herm}(\mathcal{H})$.
    \end{enumerate}
    The \textbf{cost function} is the expectation value functional:
    \begin{equation}
        \mathcal{L}(\theta) = \langle\psi_0| U^\dagger(\theta) O U(\theta) |\psi_0\rangle = \mathrm{Tr}(\rho_0 U^\dagger(\theta) O U(\theta))
    \end{equation}
    where $\rho_0 = |\psi_0\rangle\langle\psi_0|$. For ground-state preparation, we adopt the cost operator:
    \begin{equation}
        O = I - \frac{1}{n} \sum_{i=1}^n Z_i
    \end{equation}
    where $Z_i \in \mathrm{End}(\mathcal{H})$ denotes the Pauli-Z operator on the $i$-th tensor factor.
\end{frame}

% Slide 3: Barren Plateau & Master Formula
\begin{frame}{The Barren Plateau \& Master Formula}
    \textbf{The Barren Plateau Phenomenon:} Gradients vanish exponentially with system size:
    \begin{equation}
        \mathrm{Var}_{\theta}\!\left[\frac{\partial\mathcal{L}}{\partial\theta_k}\right] \le \mathcal{O}(b^{-n}), \quad b>1
    \end{equation}

    \textbf{The Lie Algebraic Master Formula:}\footnote{T. S. Pratheek (2026), M. Larocca et al. (2025)}
    For $U$ sampled from a 2-design over the Dynamical Lie Algebra (DLA) target space $U(d)$, and traceless operators $H = \rho_0 - I/d$ and $V = O - \mathrm{Tr}(O)/d \cdot I$:
    \begin{equation}
        \mathrm{Var}[\mathcal{L}] \approx \frac{\mathrm{Tr}(H^2)\,\mathrm{Tr}(V^2)}{d^2 - 1}
    \end{equation}
    \vspace{0.5em}
    \textit{Insight:} The gradient variance is strictly governed by the Hilbert–Schmidt trace norms of the initial state ($H$) and the measurement operator ($V$) relative to the Haar integration volume ($d^2$).
\end{frame}

% Slide 4: Q-LINK Extended Space
\begin{frame}{The Q-LINK Extended Hilbert Space}
    \textbf{Definition 1.6 (Extended Hilbert Space).} The Q-LINK architecture appends a single \textbf{messenger qubit}, $\mathcal{H}_m = \mathbb{C}^2$, to the data register:
    \begin{equation}
        \mathcal{H}_{\mathrm{tot}} = \mathcal{H} \otimes \mathcal{H}_m, \quad \dim(\mathcal{H}_{\mathrm{tot}}) = 2N = 2^{n+1}
    \end{equation}
    The global initial state is $\rho_{\mathrm{tot}} = \rho_0 \otimes |0\rangle\langle0|_m$. The cost operator acts as $\tilde{O} = O \otimes I_m$, measuring only the data qubits. A single $H$ gate is applied to the messenger at $t=0$ as part of $U(\theta)$.

    \vspace{1em}
    \textbf{Architectural Motivation:} The messenger qubit, initialized to $|0\rangle_m$ and never measured, acts as a \textbf{coherent residual channel}. It collects phase information from all data qubits before each operation layer and redistributes it after.

    \textit{Crucially, this pathway preserves unitarity and avoids measurement-induced decoherence — the central limitation of prior classical residual approaches (e.g.\ ResQNet).}
\end{frame}

% Slide 6: DLA Equivalence
\begin{frame}{Dynamical Lie Algebra (DLA) Equivalence}
    \small
    \textbf{Theorem 5.2:} Both the Vanilla architecture on $n+1$ qubits and Q-LINK on $n+1$ total qubits generate the full special unitary group: $\mathfrak{g}_{\mathrm{Q\text{-}LINK}} = \mathfrak{g}_{\mathrm{Vanilla}} = \mathfrak{su}(2^{n+1})$.

    \textbf{Proof (Vanilla):} Single-qubit Pauli generators $\{i X_k, i Y_k, i Z_k\}_{k=1}^{n+1}$ together with the CZ entangler $\{i\, CZ_{j,j+1} = \tfrac{i}{4}(I-Z_j)(I-Z_{j+1})\}$ form a connected entangling set on a linear graph. By the universality criterion for connected qubit graphs, the closure generates $\mathfrak{su}(2^{n+1})$.

    \textbf{Proof (Q-LINK):} The data-register subalgebra spanned by the rotations and the CZ chain equals $\mathfrak{su}(2^n)\otimes I_m \subsetneq \mathfrak{su}(2^{n+1})$. We extend by exhibiting $i\,Z_m \in \mathfrak{g}_{\mathrm{Q\text{-}LINK}}$:
    \begin{itemize}
        \item The $CNOT_{m \to i}$ in the distribution block decomposes as $\tfrac{1}{2}(I\!+\!Z_m)\otimes I_i + \tfrac{1}{2}(I\!-\!Z_m)\otimes X_i$, whose Hermitian generator contains the term $i Z_m \otimes I$ after Hadamard conjugation on $X_i$.
        \item Once $i Z_m$ is in the algebra, $[i Z_m \otimes I, \, i\,(X_m\otimes X_i)] = -2 i\, Y_m \otimes X_i$. Together with the seed $i\, X_m \otimes X_i$ from $R_{xx}$, all messenger Paulis $X_m, Y_m, Z_m$ couple to every data-qubit Pauli; iterated commutators with data-data CZs span all Pauli tensor products on $n+1$ qubits.
    \end{itemize}

    \textbf{Remark 5.3:} Expressibility is identical at depth saturation. The performance advantage is entirely attributable to the modified trace structure $\mathrm{Tr}(\tilde V^2)$.
\end{frame}

% Slide 7: Gradient Variance via Parameter-Shift Rule
\begin{frame}{Gradient Variance via Parameter-Shift Rule}
    To evaluate the Barren Plateau, we must bound the gradient variance $\mathrm{Var}[\partial\mathcal{L}/\partial\theta_k]$.

    \textbf{The Parameter-Shift Rule:}
    \begin{equation}
        \frac{\partial\mathcal{L}}{\partial\theta_k} = \frac{1}{2}\!\left[ \mathcal{L}\!\left(\theta + \tfrac{\pi}{2}e_k\right) - \mathcal{L}\!\left(\theta - \tfrac{\pi}{2}e_k\right) \right]
    \end{equation}

    Taking the variance over the Haar-saturated ensemble and using the 2-design property to cancel the $L^+$–$L^-$ covariance:
    \begin{equation}
        \mathrm{Var}\!\left[\tfrac{\partial\mathcal{L}}{\partial\theta_k}\right] = \tfrac{1}{4}\bigl(\mathrm{Var}[\mathcal{L}^+] + \mathrm{Var}[\mathcal{L}^-]\bigr) \le \tfrac{1}{2}\, \mathrm{Var}[\mathcal{L}]
    \end{equation}

    \textbf{Theorem 4.1 Consequence:} Because the parameter-shift gradient is linear in $\mathcal{L}$, it inherits the Haar variance of the cost function up to the constant $\tfrac{1}{2}$. This constant is common to both architectures and cancels in the \emph{ensemble} ratio. Therefore the architectural advantage is the ratio of $\mathrm{Var}[\mathcal{L}]$ — at the Haar 2-design limit.

    \textit{Caveat.} The empirical protocol below measures \emph{trajectory} variance during SGD, a finite-sample dynamical quantity that need not equal the Haar ensemble variance. The two coincide only at saturation.
\end{frame}

% Slide 8: Master Formula Application & Asymptotic Ratio
\begin{frame}{Master Formula Application \& Asymptotic Ratio}
    Evaluating $\mathrm{Var}[\mathcal{L}] = \mathrm{Tr}(H^2)\mathrm{Tr}(V^2)/(d^2-1)$ for identical extended dimensions $d = 2N = 2^{n+1}$.

    \textbf{Initial State Purity:} Both architectures initialize a pure tensor-product state $\rho_{\mathrm{tot}} = \rho_0 \otimes |0\rangle\langle 0|_m$:
    \begin{equation}
        H = \rho_{\mathrm{tot}} - \tfrac{I}{2N} \quad\Longrightarrow\quad \mathrm{Tr}(H^2) = \mathrm{Tr}(\rho_{\mathrm{tot}}^2) - \tfrac{1}{2N} = 1 - \tfrac{1}{2N}
    \end{equation}
    Because $d$ and $\mathrm{Tr}(H^2)$ are strictly identical, the relative gradient variance reduces purely to the ratio of the Hilbert–Schmidt cost-operator traces:
    \begin{equation}
        \mathcal{R}(n) = \frac{\mathrm{Var}_{\mathrm{Q\text{-}LINK}}[\mathcal{L}]}{\mathrm{Var}_{\mathrm{Vanilla}}[\mathcal{L}]} = \frac{\mathrm{Tr}(\tilde V^2)}{\mathrm{Tr}(V_{\mathrm{Vanilla},n+1}^2)} = \frac{2N/n}{2N/(n+1)} = \frac{n+1}{n}
    \end{equation}
    {\footnotesize \textit{Notation:} $n = n_{\mathrm{data}}$. Equivalently $\mathcal{R} = n_{\mathrm{tot}}/(n_{\mathrm{tot}}-1)$.}

    \textbf{Remark 4.2 (Asymptotic Triviality):} $\mathcal{R}(n) \to 1$ as $n \to \infty$. The Haar-saturated structural advantage vanishes. Any empirically larger advantage at finite depth $D = n^2 \log n$ comes from the Cheeger amplification factor $\mathcal{C}(n,D)$ — \textit{not} from the static ratio.
\end{frame}

% Slide 9: The actual TC bug (honest, corrected statement)
\begin{frame}{The TensorCircuit Cost Function — What the Bug Actually Does}
    \small
    The authors' \texttt{src/qlink/metrics/cost\_function.py} iterates
    \[
        \texttt{for j in range(0, 2**num\_qubits)}
    \]
    over a probability vector of length $2^{n_{\mathrm{tot}}}$, where \texttt{num\_qubits} has been decremented to $n_{\mathrm{data}} = n_{\mathrm{tot}}-1$ for Q-LINK. Half the probability vector is dropped.

    \textbf{What that actually projects.} TC is big-endian (qubit~0 is MSB; verified by\\
    \texttt{tc.Circuit(2); X(0)} $\to$ \texttt{[0,0,1,0]}). Restricting $j \in [0, 2^{n_{\mathrm{data}}})$ forces the MSB of the full $n_{\mathrm{tot}}$-bit index to $0$, i.e. it projects the \textbf{first data qubit $q_0$ onto $|0\rangle$} — \emph{not} the messenger.\\
    Inside the truncated string, the bit indexed by $i$ then refers to qubits $1,\dots,n_{\mathrm{tot}}-1$, which excludes $q_0$ from the measurement and silently includes the messenger as one of the ``measured'' slots.

    \textbf{Operator form (one line):}
    \[
        \boxed{\;\hat O_{\mathrm{buggy}} \;=\; I \;-\; \frac{1}{n_{\mathrm{data}}}\sum_{k=1}^{n_{\mathrm{tot}}-1} P_0^{(q_0)} P_0^{(q_k)}\;}
    \]
    \textbf{Consequences.} (i) Vanilla rows are \emph{unaffected} ($n_{\mathrm{tot}} = n_{\mathrm{data}}$, loop bound matches). (ii) The Q-LINK trajectory variance is inflated by the projector $P_0^{(q_0)}$, which conditions the measurement on $q_0 = |0\rangle$. (iii) The static Haar ratio $(n+1)/n$ of Slide 8 holds for the \emph{corrected} cost $\hat O_{\mathrm{fixed}} = I - \tfrac{1}{n_{\mathrm{data}}}\sum_i Z_i$.

    Unit test pinning $\hat O_{\mathrm{buggy}}$ to the authors' literal Python loop ships with the replication: \texttt{tests/unit/test\_observables.py::test\_buggy\_observable\_matches\_literal\_loop}.
\end{frame}

% Slide 13: Theoretical Extensions (E1-E5)
\begin{frame}{Theoretical Extensions (E1 -- E5)}
    \begin{itemize}
        \item \textbf{E1. Multi-Messenger Scaling ($k>1$):} \textit{Pruned.} Yields a catastrophic $\mathcal{O}(2^{-k})$ Haar volume penalty that dominates the linear trace advantage.
        \item \textbf{E2. Interaction Basis Ablation:} \textit{Constraint.} $R_{yy}$ preserves the algebraic isomorphism. $R_{zz}$ trivially commutes with the distribution controls, fracturing the DLA.
        \item \textbf{E3. Mechanism Ablation:} \textit{Pruned.} The topology is irreducible. Ablating $U_{\mathrm{coll}}$ or $U_{\mathrm{dist}}$ independently triggers algebraic stagnation $\mathfrak{su}(2^n) \oplus \mathfrak{su}(2^n)$.
        \item \textbf{E4. Periodic Temporal Sparsity:} \textit{Functional.} The continuous DLA is invariant under discrete $p$-periodicity. This dynamically throttles operator spreading.
        \item \textbf{E5. Combinatorial Cost Scaling:} \textit{Breakthrough.} Normalizing an $m$-local observable amplifies the trace ratio to $\mathcal{R}_m(n) = \tfrac{n+1}{n+1-m}$.
    \end{itemize}
\end{frame}
